def apply_orientation(img, ori):
    """与前端 rcApplyOrientation 的 setTransform 等价。"""
    if ori == 2:
        return cv2.flip(img, 1)
    if ori == 3:
        return cv2.flip(img, -1)
    if ori == 4:
        return cv2.flip(img, 0)
    if ori == 5:
        return cv2.transpose(img)
    if ori == 6:
        return cv2.rotate(img, cv2.ROTATE_90_CLOCKWISE)
    if ori == 7:
        return cv2.flip(cv2.transpose(img), -1)
    if ori == 8:
        return cv2.rotate(img, cv2.ROTATE_90_COUNTERCLOCKWISE)
    return img

def angle_between(a, b):
    v1 = (a[0] - b[0], a[1] - b[1])
    v2 = (a[2] - b[2], a[3] - b[3])
    n1, n2 = math.hypot(*v1), math.hypot(*v2)
    if n1 < 1e-6 or n2 < 1e-6:
        return 0.0
    c = max(-1.0, min(1.0, (v1[0] * v2[0] + v1[1] * v2[1]) / (n1 * n2)))
    return math.degrees(math.acos(c))


def judge(corners, W, H):
    """规则判定，不做生成式补图；返回 (status, reasons)。"""
    p = [tuple(c) for c in corners]
    top, bottom = math.dist(p[0], p[1]), math.dist(p[3], p[2])
    left, right = math.dist(p[0], p[3]), math.dist(p[1], p[2])
    reasons = []
    if min(top, bottom) < 1e-6 or min(left, right) < 1e-6:
        return '待定 · 角点退化', ['边长接近 0']
    r_h, r_v = top / bottom, left / right
    if r_h < 0.5 or r_h > 2.0:
        reasons.append('上下边长度比 %.2f（斜拍拉伸严重）' % r_h)
    if r_v < 0.5 or r_v > 2.0:
        reasons.append('左右边长度比 %.2f（斜拍拉伸严重）' % r_v)
    # 四角是否接近矩形：相邻边夹角偏离 90° 的程度
    devs = []
    for i in range(4):
        a, b, c = p[i], p[(i + 1) % 4], p[(i - 1) % 4]
        v1 = (b[0] - a[0], b[1] - a[1])
        v2 = (c[0] - a[0], c[1] - a[1])
        n1, n2 = math.hypot(*v1), math.hypot(*v2)
        if n1 < 1e-6 or n2 < 1e-6:
            devs.append(90)
            continue
        cosv = (v1[0] * v2[0] + v1[1] * v2[1]) / (n1 * n2)
        devs.append(abs(90 - math.degrees(math.acos(max(-1, min(1, cosv))))))
    if max(devs) > 45:
        reasons.append('四边形内角偏离直角最大 %.1f°' % max(devs))
    Hm = cv2.getPerspectiveTransform(np.float32(corners),
                                    np.float32([[0, 0], [W - 1, 0], [W - 1, H - 1], [0, H - 1]]))
    cond = float(np.linalg.cond(Hm))
    if cond > 1e7:
        reasons.append('单应矩阵条件数 %.2e（数值不稳定）' % cond)
    return ('待定 · 需人工复核' if reasons else '可用'), reasons


def rectify(src_path, meta, out_dir, k1_override=None, dry=False):
    name = meta.get('source_file') or src_path.name
    corners = meta.get('corners_px')
    if not corners or len(corners) != 4:
        log('  跳过 %s：JSON 缺少 4 个角点' % name)
        return None
    W, H = meta.get('output_size', [0, 0])
    if not W or not H:
        log('  跳过 %s：JSON 缺少 output_size' % name)
        return None
    ori = int(meta.get('exif_orientation', 1) or 1)
    work = tuple(meta.get('working_image_size') or [0, 0])
    k1 = float(k1_override if k1_override is not None else meta.get('lens_k1', 0) or 0)
    status, reasons = judge(corners, W, H)

    if dry:
        log('  [dry-run] %s -> %dx%d  k1=%.4f  ori=%d  状态=%s %s'
            % (name, W, H, k1, ori, status, '; '.join(reasons)))
        return dict(name=name, status=status)

    img = cv2.imread(str(src_path), cv2.IMREAD_COLOR)
    if img is None:
        log('  跳过 %s：无法读取（可能不是有效图片）' % name)
        return None
    img = apply_orientation(img, ori)
    if work[0] and work[1] and (img.shape[1], img.shape[0]) != work:
        img = cv2.resize(img, work, interpolation=cv2.INTER_AREA)

    # 与前端完全一致：目标像素 -> Hinv -> 源坐标 -> 可选径向 k1 -> 双线性采样
    Hm = cv2.getPerspectiveTransform(np.float32(corners),
                                    np.float32([[0, 0], [W - 1, 0], [W - 1, H - 1], [0, H - 1]]))
    Hinv = np.linalg.inv(Hm)
    u, v = np.meshgrid(np.arange(W, dtype=np.float32), np.arange(H, dtype=np.float32))
    a = Hinv[0, 0] * u + Hinv[0, 1] * v + Hinv[0, 2]
    b = Hinv[1, 0] * u + Hinv[1, 1] * v + Hinv[1, 2]
    c = Hinv[2, 0] * u + Hinv[2, 1] * v + Hinv[2, 2]
    c = np.where(np.abs(c) < 1e-9, 1e-9, c)
    mx, my = a / c, b / c
    if k1:
        cx, cy, hx, hy = img.shape[1] / 2, img.shape[0] / 2, img.shape[1] / 2, img.shape[0] / 2
        nx, ny = (mx - cx) / hx, (my - cy) / hy
        f = 1 + k1 * (nx * nx + ny * ny)
        mx, my = cx + nx * f * hx, cy + ny * f * hy
    out = cv2.remap(img, mx, my, cv2.INTER_LINEAR,
                    borderMode=cv2.BORDER_CONSTANT, borderValue=(255, 255, 255))

    # 图外白色占比过高 -> 说明原图覆盖面不足，标待定
    white = float(np.mean(np.all(out > 248, axis=2)))
    if white > 0.25:
        status = '待定 · 需人工复核'
        reasons.append('图外白色填充占比 %.1f%%（原图覆盖不足或大面积遮挡）' % (white * 100))

    base = Path(name).stem + '_rect'
    out_dir.mkdir(parents=True, exist_ok=True)
    jpg = out_dir / (base + '.jpg')
    cv2.imwrite(str(jpg), out, [int(cv2.IMWRITE_JPEG_QUALITY), 92])
    rec = dict(meta)
    rec.update({
        'offline_batch': {
            'tool': 'batch_rectify.py', 'executed_at': datetime.datetime.now().isoformat(timespec='seconds'),
            'source_path': str(src_path.relative_to(ROOT)).replace('\\', '/'),
            'k1_applied': k1, 'outside_fill': '#ffffff（白色空白，非补图）',
            'outside_white_ratio': round(white, 4),
            'status': status, 'status_reasons': reasons,
            'homography_cv2_src_to_dst': Hm.tolist(),
            'homography_cv2_dst_to_src': Hinv.tolist(),
            'note': '变换方式与前端工具一致（四点单应 + 逆映射双线性 + 可选 k1）；原图未被修改',
        },
        'academic_note': '校正图仅用于构件标注与展示，不是测量级正射影像；论文方法章节需写明变换方式与参数；不做生成式补图',
    })
    (out_dir / (base + '.json')).write_text(json.dumps(rec, ensure_ascii=False, indent=2), encoding='utf-8')
    log('  %s  %dx%d  k1=%.4f  %s  %s'
        % (jpg.name, W, H, k1, status, ('; '.join(reasons)) if reasons else ''))
    return dict(name=name, status=status, out=str(jpg))


def auto_rectifying_homography(img):
    """LSD 线段 → 两个消失点 → 消失线 → 射影(仿射)矫正矩阵。

    H 的第三行取消失线 l，则 l 上的点满足 x'_3 = l·x = 0，即消失线被推到无穷远，
    画面中原本 converges 的水平/垂直线恢复平行。只做仿射矫正、不做度量矫正，
    结果一律标「待定 · 实验性」，必须人工在前端工具里点四角复核。
    """
    h, w = img.shape[:2]
    g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    raw = cv2.createLineSegmentDetector(cv2.LSD_REFINE_ADV).detect(g)[0]
    if raw is None:
        return None, 'LSD 未检测到线段'
    # OpenCV 4.x 返回 (N,1,4)，5.x 返回 (N,4)，统一成 (N,4)
    lines = np.asarray(raw).reshape(-1, 4)
    if len(lines) < 8:
        return None, 'LSD 线段不足，无法估计消失点'
    horiz, vert = [], []
    for L in lines:
        x1, y1, x2, y2 = map(float, L)
        if math.hypot(x2 - x1, y2 - y1) < min(w, h) * 0.07:
            continue
        ang = abs(math.degrees(math.atan2(y2 - y1, x2 - x1)))
        ang = min(ang, 180 - ang)
        if ang < 25:
            horiz.append((x1, y1, x2, y2))
        elif ang > 65:
            vert.append((x1, y1, x2, y2))
    if len(horiz) < 4 or len(vert) < 4:
        return None, '水平/垂直线段不足（h=%d, v=%d）' % (len(horiz), len(vert))

    def vp(seg):
        """RANSAC 投票求消失点：建筑立面里混入大量装饰线，直接最小二乘会被拉偏。"""
        norm = []
        for x1, y1, x2, y2 in seg:
            a, b, c = y1 - y2, x2 - x1, x1 * y2 - x2 * y1
            n = math.hypot(a, b)
            if n > 1e-9:
                norm.append((a / n, b / n, c / n))
        if len(norm) < 4:
            return None
        rng = random.Random(20261107)
        n = len(norm)
        thresh = max(2.5, min(w, h) * 0.004)
        best, best_inl = None, []
        pairs = [(i, j) for i in range(n) for j in range(i + 1, n)]
        rng.shuffle(pairs)
        for i, j in pairs[:400]:
            a1, b1, c1 = norm[i]; a2, b2, c2 = norm[j]
            d = a1 * b2 - a2 * b1
            if abs(d) < 1e-6:
                continue
            px, py = (b1 * c2 - b2 * c1) / d, (c1 * a2 - c2 * a1) / d
            if abs(px) > 1e6 or abs(py) > 1e6:
                continue
            inl = [k for k, (a, b, c) in enumerate(norm) if abs(a * px + b * py + c) < thresh]
            if len(inl) > len(best_inl):
                best_inl, best = inl, (px, py)
        if not best_inl or len(best_inl) < 4:
            return None
        A = np.array([[norm[k][0], norm[k][1]] for k in best_inl])
        y = np.array([-norm[k][2] for k in best_inl])
        v = np.linalg.lstsq(A, y, rcond=None)[0]
        return float(v[0]), float(v[1])

    vh, vv = vp(horiz), vp(vert)
    if not vh or not vv:
        return None, '消失点估计失败'
    if any(abs(c) > 20 * max(w, h) for c in vh + vv):
        return None, '消失点距离异常（近似正视图，无需矫正）'
    l = np.cross(np.array([vh[0], vh[1], 1.0]), np.array([vv[0], vv[1], 1.0]))
    if abs(l[2]) < 1e-9:
        return None, '消失线退化'
    l = l / l[2]
    quad = [(0, 0), (w, 0), (w, h), (0, h)]
    denoms = [l[0] * x + l[1] * y + 1.0 for x, y in quad]
    if min(denoms) < 0.15:
        return None, '消失线穿过画面，估计不可靠'
    H = np.array([[1, 0, 0], [0, 1, 0], [l[0], l[1], 1]], dtype=float)
    moved = []
    for x, y in quad:
        v = H @ np.array([x, y, 1.0])
        moved.append((v[0] / v[2], v[1] / v[2]))
    minx, miny = min(p[0] for p in moved), min(p[1] for p in moved)
    maxx, maxy = max(p[0] for p in moved), max(p[1] for p in moved)
    ow, oh = maxx - minx, maxy - miny
    if not (0.3 * w < ow < 3.0 * w and 0.3 * h < oh < 3.0 * h):
        return None, '矫正后尺度异常（%.0f×%.0f）' % (ow, oh)
    if not (0.35 < ow / oh < 3.0):
        return None, '矫正后长宽比异常（%.0f×%.0f）' % (ow, oh)
    T = np.array([[1, 0, -minx], [0, 1, -miny], [0, 0, 1]], dtype=float)
    Ha = T @ H
    info = dict(mode='vanishing', H=Ha, out_w=int(round(ow)), out_h=int(round(oh)),
                vh=list(vh), vv=list(vv), vanishing_line=l.tolist(),
                lines_h=len(horiz), lines_v=len(vert))
    return info, '消失点仿射矫正 h=(%.0f,%.0f) v=(%.0f,%.0f) → %d×%d' % (vh + vv + (info['out_w'], info['out_h']))


def auto_quad_plan(img):
    """消失点法失败时的回退：用最长的上/下/左/右结构线求交点，得到候选四角。
    全部结果仍标「待定 · 实验性」，必须人工在前端工具里复核四角。"""
    h, w = img.shape[:2]
    g = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    # HoughLinesP 比 LSD 更能找到跨越整幅立面的长结构线（可跨 20+ px 断口）
    edges = cv2.Canny(cv2.GaussianBlur(g, (5, 5), 0), 50, 150)
    raw = cv2.HoughLinesP(edges, 1, np.pi / 180,
                          threshold=int(min(w, h) * 0.12),
                          minLineLength=max(w, h) * 0.14,
                          maxLineGap=max(w, h) * 0.035)
    if raw is None:
        return None, 'Hough 未检测到长直线'
    lines = np.asarray(raw).reshape(-1, 4)
    H, V = [], []
    for L in lines:
        x1, y1, x2, y2 = map(float, L)
        ang = abs(math.degrees(math.atan2(y2 - y1, x2 - x1)))
        ang = min(ang, 180 - ang)
        if ang < 28:
            H.append((x1, y1, x2, y2, (y1 + y2) / 2))
        elif ang > 62:
            V.append((x1, y1, x2, y2, (x1 + x2) / 2))
    if len(H) < 2 or len(V) < 2:
        return None, '长结构线不足（h=%d, v=%d）' % (len(H), len(V))
    top, bot = min(H, key=lambda t: t[4]), max(H, key=lambda t: t[4])
    left, right = min(V, key=lambda t: t[4]), max(V, key=lambda t: t[4])

    def inter(A, B):
        x1, y1, x2, y2 = A[:4]; x3, y3, x4, y4 = B[:4]
        d = (x1 - x2) * (y3 - y4) - (y1 - y2) * (x3 - x4)
        if abs(d) < 1e-6:
            return None
        return [((x1 * y2 - y1 * x2) * (x3 - x4) - (x1 - x2) * (x3 * y4 - y3 * x4)) / d,
                ((x1 * y2 - y1 * x2) * (y3 - y4) - (y1 - y2) * (x3 * y4 - y3 * x4)) / d]

    corners = [inter(top, left), inter(top, right), inter(bot, right), inter(bot, left)]
    if any(p is None for p in corners):
        return None, '四角交点退化'
    if any(p[0] < -0.3 * w or p[0] > 1.3 * w or p[1] < -0.3 * h or p[1] > 1.3 * h for p in corners):
        return None, '候选四角超出合理范围'
    area = abs(sum(corners[i][0] * corners[(i + 1) % 4][1] -
                   corners[(i + 1) % 4][0] * corners[i][1] for i in range(4))) / 2
    if area < 0.22 * w * h:
        return None, '候选四边形面积过小'
    ow = int(round((math.dist(corners[0], corners[1]) + math.dist(corners[3], corners[2])) / 2))
    oh = int(round((math.dist(corners[0], corners[3]) + math.dist(corners[1], corners[2])) / 2))
    if not (0.3 * w < ow < 2.5 * w and 0.3 * h < oh < 2.5 * h):
        return None, '候选输出尺寸异常'
    st, reasons = judge(corners, ow, oh)
    if st != '可用':
        return None, '候选四角不合理：' + '；'.join(reasons)[:70]
    return dict(mode='quad', corners=corners, out_w=ow, out_h=oh,
                area_ratio=round(area / (w * h), 3)), '自动四角候选 → %d×%d（面积占比 %.0f%%）' % (ow, oh, 100 * area / (w * h))


def load_work_image(src_path, ori, work_side=1600):
    """读原图 → EXIF 转正 → 等比缩到长边 ≤work_side（LSD 提速），保持两处调用完全一致。"""
    img = cv2.imread(str(src_path), cv2.IMREAD_COLOR)
    if img is None:
        return None, 1.0
    img = apply_orientation(img, ori)
    h0, w0 = img.shape[:2]
    sc = min(1.0, work_side / max(h0, w0))
    if sc < 1:
        img = cv2.resize(img, (int(round(w0 * sc)), int(round(h0 * sc))),
                         interpolation=cv2.INTER_AREA)

def parse_orientation(tiff):
    """tiff 为 TIFF header 起始处的字节（Exif\\0\\0 之后）。"""
    if len(tiff) < 8:
        return 1
    if tiff[:2] == b'II':
        bo = 'little'
    elif tiff[:2] == b'MM':
        bo = 'big'
    else:
        return 1
    if int.from_bytes(tiff[2:4], bo) != 0x2A:
        return 1
    ifd = int.from_bytes(tiff[4:8], bo)
    if ifd + 2 > len(tiff):
        return 1
    cnt = int.from_bytes(tiff[ifd:ifd + 2], bo)
    for k in range(min(cnt, 512)):
        e = ifd + 2 + k * 12
        if e + 12 > len(tiff):
            break
        if int.from_bytes(tiff[e:e + 2], bo) == 0x0112:
            return int.from_bytes(tiff[e + 8:e + 10], bo) or 1
    return 1


def image_header(path):
    """按 JPEG 段 seek 读取，返回 (w, h, exif_orientation)。只读极少量字节。"""
    with open(path, 'rb') as f:
        head = f.read(8)
        if head[:8] == b'\x89PNG\r\n\x1a\n':
            f.seek(16)
            return (int.from_bytes(f.read(4), 'big'),
                    int.from_bytes(f.read(4), 'big'), 1)
        if head[:2] != b'\xff\xd8':
            return 0, 0, 1
        f.seek(2)
        w = h = 0
        ori = 1
        while True:
            b = f.read(1)
            if not b:
                break
            if b != b'\xff':
                continue                       # 填充字节，继续找 FF
            m = f.read(1)
            while m == b'\xff':                # FF FF ... 填充
                m = f.read(1)
            if not m:
                break
            mark = m[0]
            if mark in (0xD8, 0x01) or 0xD0 <= mark <= 0xD9:
                continue
            lb = f.read(2)
            if len(lb) < 2:
                break
            L = int.from_bytes(lb, 'big')
            if L < 2:
                break
            if mark in SOF:
                seg = f.read(5)
                if len(seg) >= 5:
                    h = int.from_bytes(seg[1:3], 'big')
                    w = int.from_bytes(seg[3:5], 'big')
            elif mark == 0xE1 and L < 2_000_000:
                seg = f.read(L - 2)
                if seg[:4] == b'Exif':
                    ori = parse_orientation(seg[6:])
            else:
                f.seek(L - 2, 1)
            if mark == 0xDA or (w and ori != 1):
                # SOS 之后是压缩数据；若已拿到尺寸且方向非 1 可提前退出
                if mark == 0xDA:
                    break
            elif w and mark in SOF:
                # SOF 已取到，但还要继续找 EXIF？EXIF 一定在 SOF 之前，可退出
                break
    return w, h, ori


def md5(path):
    h = hashlib.md5()
    with open(path, 'rb') as f:
        for blk in iter(lambda: f.read(1 << 20), b''):
            h.update(blk)
    return h.hexdigest()

"""

def read_matrix(path, min_freq):
    with open(path, encoding="utf-8-sig", newline="") as f:
        rows = list(csv.DictReader(f))
    if not rows:
        raise SystemExit("矩阵为空：请至少填写一行 0/1 数据")
    headers = list(rows[0].keys())
    # 构件列 = 非元信息列，且所有取值都属于 {'', '0', '1'}
    candidates = [h for h in headers if h not in META_COLS and h.strip()]
    elem_cols = [
        h for h in candidates
        if all((r.get(h) or "").strip() in ("", "0", "1") for r in rows)
    ]
    if not elem_cols:
        raise SystemExit("没找到构件列：请检查表头，构件列取值只能是 0 / 1 / 空")

    freq = {

        for c in elem_cols
    }
    dropped = [c for c in elem_cols if freq[c] < min_freq] if min_freq > 0 else []
    used = [c for c in elem_cols if c not in dropped]

    data = []
    for r in rows:
        vec = tuple(1 if (r.get(c) or "").strip() == "1" else 0 for c in used)
        if not any(vec):
            continue  # 全 0 串，对应论文的 000...0 过滤
        name = (r.get("样本编号") or "?").strip() or "?"
        data.append((name, vec))
    return used, freq, dropped, data


def print_strings(data):
    counter = Counter(v for _, v in data)
    return ["".join(map(str, v)) for v in counter], counter


def seriation(vecs):
    """Wilkinson 无约束序列重排：交替按行质心、列质心排序至稳定。"""
    n, m = len(vecs), len(vecs[0])
    rows, cols = list(range(n)), list(range(m))
    for _ in range(200):
        def row_centroid(i):
            pos = [p + 1 for p, c in enumerate(cols) if vecs[i][c] == 1]
            return (sum(pos) / len(pos) if pos else m + 1, i)

        new_rows = sorted(rows, key=row_centroid)

        def col_centroid(j):
            pos = [p + 1 for p, i in enumerate(new_rows) if vecs[i][j] == 1]
            return (sum(pos) / len(pos) if pos else n + 1, j)

        new_cols = sorted(cols, key=col_centroid)
        if new_rows == rows and new_cols == cols:
            break
        rows, cols = new_rows, new_cols
    return rows, cols


def pdistance(a, b):
    m = len(a)
    return sum(1 for x, y in zip(a, b) if x != y) / m if m else 0.0


def neighbor_joining(names, vecs):
    """标准 NJ：Q 判据选对 -> 记枝长 -> 合并 -> 输出 Newick。"""
    n = len(names)
    if n < 2:
        return (names[0] + ";" if n else ";")
    m = len(vecs[0])
    cur = [[0.0] * n for _ in range(n)]
    for i in range(n):
        for j in range(i + 1, n):
            d = pdistance(vecs[i], vecs[j])
            cur[i][j] = cur[j][i] = d
    ids = list(range(n))
    labels = {i: names[i] for i in range(n)}
    nxt = n
    while len(ids) > 2:
        r = len(ids)
        sums = [sum(cur[i]) for i in range(r)]
        best = None
        for i in range(r):
            for j in range(i + 1, r):
                q = (r - 2) * cur[i][j] - sums[i] - sums[j]
                if best is None or q < best[0]:
                    best = (q, i, j)
        _, i, j = best
        dij = cur[i][j]
        ri = sums[i] / (r - 2) if r > 2 else 0.0
        rj = sums[j] / (r - 2) if r > 2 else 0.0
        li = max(0.0, 0.5 * dij + 0.5 * (ri - rj))
        lj = max(0.0, dij - li)
        new = nxt
        nxt += 1
        labels[new] = "(%s:%.5f,%s:%.5f)" % (labels[ids[i]], li, labels[ids[j]], lj)
        keep = [k for k in range(r) if k not in (i, j)]
        new_row = [max(0.0, 0.5 * (cur[i][k] + cur[j][k] - dij)) for k in keep]
        size = len(keep) + 1
        new_cur = [[0.0] * size for _ in range(size)]
        for a, ka in enumerate(keep):
            for b, kb in enumerate(keep):
                new_cur[a][b] = cur[ka][kb]
            new_cur[a][size - 1] = new_cur[size - 1][a] = new_row[a]
        ids = [ids[k] for k in keep] + [new]
        cur = new_cur
    d = cur[0][1]
    return "(%s:%.5f,%s:%.5f);" % (labels[ids[0]], d / 2, labels[ids[1]], d / 2)

class QuietHandler(http.server.SimpleHTTPRequestHandler):
    """静态文件 + /ollama/* 纯转发（无业务逻辑，不做任何图像处理）。"""

    def _cors(self):
        self.send_header("Access-Control-Allow-Origin", "*")
        self.send_header("Access-Control-Allow-Methods", "GET, POST, OPTIONS")
        self.send_header("Access-Control-Allow-Headers", "Content-Type")

        self.send_header("Content-Length", "0")

    def _proxy(self, method):
        if not self.path.startswith("/ollama/"):
            self.send_error(404)
            return
        target = OLLAMA_BASE + self.path[len("/ollama"):]
        length = int(self.headers.get("Content-Length") or 0)
        body = self.rfile.read(length) if length else None
        req = urllib.request.Request(target, data=body, method=method,
                                     headers={"Content-Type": "application/json"})
        try:
            with urllib.request.urlopen(req, timeout=180) as r:
                data, status = r.read(), r.status
                ctype = r.headers.get("Content-Type", "application/json")
        except urllib.error.HTTPError as e:

        except Exception as e:
            data = json.dumps({"error": str(e)}, ensure_ascii=False).encode()
            status, ctype = 502, "application/json"
        self.send_response(status)
        self.send_header("Content-Type", ctype)
        self.send_header("Content-Length", str(len(data)))
        self._cors()
        self.end_headers()
        try:
            self.wfile.write(data)
        except (BrokenPipeError, ConnectionResetError):
            pass

    def do_GET(self):
        if self.path.startswith("/ollama/"):
            return self._proxy("GET")
        return super().do_GET()

    def do_POST(self):

def load_vocabulary():
    html = (HERE / "index.html").read_text(encoding="utf-8")
    m = re.search(r"const ELEMENTS = (\[[\s\S]*?\]);\n", html)
    g = re.search(r"const GROUPS = (\[[\s\S]*?\]);\n", html)
    if not m or not g:
        raise RuntimeError("无法从 index.html 读取 ELEMENTS / GROUPS")
    elems = json.loads(m.group(1))
    groups = dict(re.findall(r"\{id:'([^']+)',\s*name:'([^']+)'", g.group(1)))

def image_size(data):
    """纯标准库读 JPEG SOF / PNG IHDR 宽高。"""
    if data[:8] == b"\x89PNG\r\n\x1a\n":
        return struct.unpack(">II", data[16:24])
    if data[:2] != b"\xff\xd8":
        return 0, 0
    i, n = 2, len(data)
    while i + 9 < n:
        if data[i] != 0xFF:
            i += 1
            continue
        m = data[i + 1]
        if m in (0xD8, 0x01) or 0xD0 <= m <= 0xD9:
            i += 2
            continue
        L = int.from_bytes(data[i + 2:i + 4], "big")
        if L < 2:
            break
        if m in (0xC0, 0xC1, 0xC2, 0xC3, 0xC5, 0xC6, 0xC7, 0xC9, 0xCA, 0xCB, 0xCD, 0xCE, 0xCF):
            return (int.from_bytes(data[i + 7:i + 9], "big"),
                    int.from_bytes(data[i + 5:i + 7], "big"))
        if m == 0xDA:
            break
        i += 2 + L
    return 0, 0


def shrink_for_vlm(data, side=640):
    """给 VLM 的缩图；有 cv2 就用，没有就原图（Ollama 会自行缩放，只是慢一点）。"""
    try:
        cp = cv_search_path()
        if cp and cp not in sys.path:
            sys.path.insert(0, cp)
        import cv2
        import numpy as np
        arr = cv2.imdecode(np.frombuffer(data, np.uint8), cv2.IMREAD_COLOR)
        if arr is None:
            return data
        h, w = arr.shape[:2]
        sc = min(1.0, side / max(h, w))
        if sc < 1:
            arr = cv2.resize(arr, (int(w * sc), int(h * sc)), interpolation=cv2.INTER_AREA)
        ok, buf = cv2.imencode(".jpg", arr, [int(cv2.IMWRITE_JPEG_QUALITY), 80])
        return buf.tobytes() if ok else data
    except Exception:
        return data


def shrink_for_annotation(data, side=1600):
    """写进 labelme JSON 的图：保持可标注分辨率，同时别让 JSON 太大。"""
    try:
        cp = cv_search_path()
        if cp and cp not in sys.path:
            sys.path.insert(0, cp)
        import cv2
        import numpy as np
        arr = cv2.imdecode(np.frombuffer(data, np.uint8), cv2.IMREAD_COLOR)
        if arr is None:
            return data, image_size(data)
        h, w = arr.shape[:2]
        sc = min(1.0, side / max(h, w))
        if sc < 1:
            arr = cv2.resize(arr, (int(w * sc), int(h * sc)), interpolation=cv2.INTER_AREA)
        ok, buf = cv2.imencode(".jpg", arr, [int(cv2.IMWRITE_JPEG_QUALITY), 88])
        if ok:
            out = buf.tobytes()
            return out, (arr.shape[1], arr.shape[0])
    except Exception:
        pass
    return data, image_size(data)

def check_ollama(model, timeout=6):
    """连接自检：返回 (ok, 说明)。分清服务没起 / 模型没装 / 正常。"""
    try:
        with urllib.request.urlopen("http://127.0.0.1:11434/api/tags", timeout=timeout) as r:
            o = json.load(r)
    except Exception as e:
        return False, "连不上本机 Ollama（127.0.0.1:11434）：%s。请先启动 Ollama 服务。" % e
    names = [m.get("name", "") for m in o.get("models", [])]

def call_ollama(b64, prompt, model, timeout=180, retry=False):
    # 2026-11-09 实测：qwen3-vl:4b 是 qwen3-vl-thinking renderer，
    # 默认上下文仅 8192；显式给 think/num_ctx/num_predict 才能稳定出结果。
    body = {"model": model, "stream": False, "think": False,
            "options": {"temperature": 0.3 if retry else 0.1,
                        "num_predict": 2048, "num_ctx": 8192},
            "messages": [{"role": "user", "content": prompt, "images": [b64]}]}
    req = urllib.request.Request(OLLAMA, data=json.dumps(body).encode(),
                                 headers={"Content-Type": "application/json"})
    with urllib.request.urlopen(req, timeout=timeout) as r:
        return json.load(r)


def extract_items(text, valid_ids=None):
    """qwen3-vl:4b 是 thinking 模型：常把完整 JSON 写在 thinking 里、content 为空，
    有时思考没写完就被截断，只剩自然语言列举 + 坐标。故做三层容错。"""
    out = []
    # ① 完整 {"items":[...]}
    for m in re.finditer(r'\{\s*"items"\s*:\s*\[([\s\S]*?)\]\s*\}', text):
        try:
            o = json.loads('{"items":[' + m.group(1) + ']}')
            if isinstance(o.get("items"), list):
                out.append(o["items"])
        except Exception:
            pass
    # ② 分散的 "label"/"bbox"/"conf" 片段
    if not out:
        one = []
        for m in re.finditer(r'"label"\s*:\s*"([a-z_]+)"\s*,\s*"bbox"\s*:\s*\[([^\]]+)\]\s*,\s*"conf"\s*:\s*([\d.]+)', text):
            bb = [float(x) for x in m.group(2).split(",")]
            if len(bb) == 4:
                one.append({"label": m.group(1), "bbox": bb, "conf": float(m.group(3))})
        if one:
            out.append(one)
    # 只有 ①② 的 JSON 才算「已有结构化结果」；③④ 都是自然语言兜底，需合并去重
    has_json = bool(out)
    # ③ 自然语言列举：「1. broken_pediment：顶部山花，坐标[100, 20, 900, 180]」
    if not has_json:
        one = []
        pattern = (r'(?:^|\n)\s*(?:\d+\s*[.、)]\s*|[-*]\s*)?'
                   r'([a-z][a-z_]{3,})\s*[:：]?\s*[^\[\]\n]{0,70}?'
                   r'\[\s*(\d{1,4})\s*,\s*(\d{1,4})\s*,\s*(\d{1,4})\s*,\s*(\d{1,4})\s*\]')
        for m in re.finditer(pattern, text):
            lab = m.group(1)
            if valid_ids is not None and lab not in valid_ids:
                continue
            # 这种来源是模型估计，置信度一律压到 0.5（且仍标草稿）
            one.append({"label": lab,
                        "bbox": [int(m.group(i)) for i in range(2, 6)],
                        "conf": 0.5})
        if one:
            out.append(one)
    # ③ 只认英文 id，可能先「抢答」而挡住 ④ 的中文名匹配；
    # 故无 JSON 时 ③④ 都跑，最后合并去重。
    # ④ thinking 里的自然语言估算：「broken_pediment：顶部装饰，所以 bbox[150, 80, 850, 180]」
    #    或「46遮阳板：底部遮阳棚，所以bbox[0, 700, 1000, 800]」。content 为空时这是唯一来源。
    if not has_json:
        one, seen_box = [], set()
        name2id = {}
        for e in ELEMENTS_CACHE:
            name2id[e["name"]] = e["id"]
            for part in re.split(r"[／/]", e["name"]):
                p = part.strip()
                if len(p) >= 2 and p not in name2id:
                    name2id[p] = e["id"]
        for m in re.finditer(r"\[\s*(\d{1,4})\s*,\s*(\d{1,4})\s*,\s*(\d{1,4})\s*,\s*(\d{1,4})\s*\]", text):
            b = [int(m.group(i)) for i in range(1, 5)]
            if b[2] <= b[0] or b[3] <= b[1]:
                continue
            bk = ",".join(map(str, b))
            if bk in seen_box:
                continue
            ctx = text[max(0, m.start() - 140):m.start()]
            lab, best_pos = None, -1
            for mm in re.finditer(r"[a-z][a-z_]{3,}", ctx):
                if (valid_ids is None or mm.group(0) in valid_ids) and mm.start() > best_pos:
                    lab, best_pos = mm.group(0), mm.start()
            for nm, eid in name2id.items():
                if len(nm) < 2:
                    continue
                p = ctx.rfind(nm)
                if p >= 0 and p + len(nm) > best_pos:
                    lab, best_pos = eid, p + len(nm)
            if lab:
                seen_box.add(bk)
                one.append({"label": lab, "bbox": b, "conf": 0.5})
        if one:
            out.append(one)
    if not has_json and len(out) > 1:
        merged, seen = [], set()
        for arr in out:
            for it in arr:
                k = str(it.get("label")) + "|" + ",".join(map(str, it.get("bbox") or []))
                if k not in seen:
                    seen.add(k)
                    merged.append(it)
        return merged
    return max(out, key=len) if out else []


def normalize(items, valid_ids, max_items):
    seen, res = set(), []
    for it in items:
        lab = (it or {}).get("label")
        if lab not in valid_ids:
            continue
        b = [float(x) for x in (it.get("bbox") or [])]
        if len(b) != 4:
            continue
        x1, y1, x2, y2 = [max(0.0, min(1000.0, v)) / 1000.0 for v in b]
        if x2 <= x1 or y2 <= y1 or (x2 - x1) > 0.995 or (y2 - y1) > 0.995:
            continue
        key = lab + "|" + ",".join("%.2f" % v for v in (x1, y1, x2, y2))
        if key in seen:
            continue
        seen.add(key)
        try:
            conf = float(it.get("conf", 0))
        except Exception:
            conf = 0.0
        res.append({"label": lab, "box": [x1, y1, x2, y2], "vlmConf": conf})
        if len(res) >= max_items:
            break
    return res


def to_labelme_json(items, img_name, img_data, W, H, model):
    now = datetime.datetime.now().isoformat(timespec="seconds")
    shapes = []
    for i, it in enumerate(items, 1):
        x1, y1, x2, y2 = it["box"]
        ext = {"id": "vlm%d" % i, "variant": "", "conf": 0.9, "origin": "待考",
               "dispute": False, "note": "", "source": "vlm", "draft": True,
               "reviewed": False, "vlmConf": it["vlmConf"], "vlmModel": model,
               "vlmAt": now, "reviewedAt": None}
        shapes.append({
            "label": it["label"],
            "points": [[round(x1 * W, 2), round(y1 * H, 2)],
                       [round(x2 * W, 2), round(y2 * H, 2)]],
            "group_id": None, "shape_type": "rectangle",
            "flags": {"dispute": False, "draft": True, "reviewed": False},
            "description": json.dumps(ext, ensure_ascii=False),
            "mask": None,
        })
    return {
        "version": "5.8.1",
        "flags": {"draft": True, "prelabel": True},
        "shapes": shapes,
        "imagePath": img_name,
        "imageData": base64.b64encode(img_data).decode(),
        "imageHeight": H, "imageWidth": W, "text": "",
        "description": ("台山骑楼 VLM 构件预标注草稿 · 由本机 %s 生成。"
                        "全部为橙色虚线草稿，必须人工逐框复核后才计入证据分与正式导出；"
                        "框位置为模型估计值、置信度为模型自评，不可复现，不得写入论文结论。" % model),
    }


def main():
    ap = argparse.ArgumentParser(description="离线 VLM 构件预标注 → labelme JSON")
    ap.add_argument("input", help="图片路径，或包含图片的目录")
    ap.add_argument("--out", default="", help="输出目录（默认与图片同目录）")
    ap.add_argument("--model", default=DEFAULT_MODEL)
    ap.add_argument("--max", type=int, default=8, help="每张最多几个构件")
    ap.add_argument("--limit", type=int, default=0, help="目录模式最多处理几张（0=全部）")
    a = ap.parse_args()

    elems, vocab = load_vocabulary()
    globals()["ELEMENTS_CACHE"] = elems
    valid = {e["id"] for e in elems}
    prompt = build_prompt(vocab, a.max)
    print("词汇表 %d 项（紧凑格式） | 模型 %s | 本机 Ollama" % (len(elems), a.model))
    ok_svc, msg = check_ollama(a.model)
    print("连接自检：", msg)
    if not ok_svc:
        return 2

    p = Path(a.input)
    if p.is_dir():
        files = sorted([x for x in p.iterdir()
                        if x.suffix.lower() in (".jpg", ".jpeg", ".png")])
        if a.limit:
            files = files[:a.limit]
    else:
        files = [p]
    if not files:
        print("没有找到图片")
        return
    outdir = Path(a.out) if a.out else None
    if outdir:
        outdir.mkdir(parents=True, exist_ok=True)

    ok = 0
    for f in files:
        t0 = time.time()
        try:
            raw = f.read_bytes()
            vlm_img = shrink_for_vlm(raw)
            ann_img, (W, H) = shrink_for_annotation(raw)
            if not W or not H:
                print("  跳过 %s：读不到尺寸" % f.name)
                continue
            o = call_ollama(base64.b64encode(vlm_img).decode(), prompt, a.model)
            msg = o.get("message", {})
            text = (msg.get("content") or "") + "\n" + (msg.get("thinking") or "")
            items = normalize(extract_items(text, valid), valid, a.max)
            if not items:
                print("    首次未取到草稿，自动重试一次…")
                retry_prompt = build_prompt(vocab, min(4, a.max), retry=True)
                o = call_ollama(base64.b64encode(vlm_img).decode(), retry_prompt,
                                a.model, retry=True)
                msg = o.get("message", {})
                text = (msg.get("content") or "") + "\n" + (msg.get("thinking") or "")
                items = normalize(extract_items(text, valid), valid, a.max)
            obj = to_labelme_json(items, f.name, ann_img, W, H, a.model)
            dest = (outdir or f.parent) / (f.stem + "_prelabel.json")
            dest.write_text(json.dumps(obj, ensure_ascii=False, indent=2), encoding="utf-8")
            ok += 1
            print("  %s → %d 个草稿框 | %.1fs | %s" %
                  (f.name[:42], len(items), time.time() - t0, dest.name))
        except Exception as e:
            print("  失败 %s：%s %s" % (f.name[:42], type(e).__name__, str(e)[:120]))
    print("完成 %d/%d 张。回到工作台点「导入 labelme」选这些 JSON，即可逐框复核。" % (ok, len(files)))


if __name__ == "__main__":
    sys.exit(main() or 0)
